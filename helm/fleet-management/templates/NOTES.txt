────────────────────────────────────────────────────────────────
📋 USEFUL COMMANDS
────────────────────────────────────────────────────────────────

View backend pods:
  kubectl get pods -n {{ .Values.namespaces.app }} -l app={{ .Values.backend.name }}

View backend logs:
  kubectl logs -f -n {{ .Values.namespaces.app }} -l app={{ .Values.backend.name }}

View database pods:
  kubectl get pods -n {{ .Values.namespaces.data }} -l app={{ .Values.postgresql.name }}

Check HPA status:
  kubectl get hpa -n {{ .Values.namespaces.app }}

Rollout status:
  kubectl rollout status deployment/{{ .Values.backend.name }} -n {{ .Values.namespaces.app }}

Restart backend:
  kubectl rollout restart deployment/{{ .Values.backend.name }} -n {{ .Values.namespaces.app }}
